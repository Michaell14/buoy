import { useEffect, useState } from 'react';
import { buoy, BuoyLive } from 'buoy-ui';

export function RecordButton() {
  const [recording, setRecording] = useState(false);
  const [seconds, setSeconds] = useState(0);

  useEffect(() => {
    if (!recording) return;
    setSeconds(0);
    const tick = setInterval(() => setSeconds((s) => s + 1), 1000);
    return () => clearInterval(tick);
  }, [recording]);

  // One way out, whether Stop was pressed here or on the island.
  function stop() {
    setRecording(false);
    buoy.success('Voice memo saved', {
      description: `${seconds} seconds`,
    });
  }

  return (
    <>
      <button onClick={recording ? stop : () => setRecording(true)}>
        {recording ? 'Stop' : 'Record'}
      </button>

      {recording && (
        <BuoyLive label="Recording a voice memo" expandable>
          {({ expanded }) =>
            expanded ? (
              <div className="buoy-exp">
                <span className="buoy-exp-title">Voice memo</span>
                <div className="buoy-actions">
                  <button className="buoy-btn is-primary" onClick={stop}>
                    Stop
                  </button>
                </div>
              </div>
            ) : (
              // The compact view sits inside a button, so
              // controls go in the expanded view.
              <span className="buoy-row no-icon">
                <span className="buoy-title">Recording</span>
                <span className="buoy-pct">{seconds} s</span>
              </span>
            )
          }
        </BuoyLive>
      )}
    </>
  );
}
