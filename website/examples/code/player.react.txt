import { BuoyLive } from 'buoy-ui';

export function Player({ track, playing, onToggle }: PlayerProps) {
  // On the island for as long as <BuoyLive> is rendered. Its
  // children are ordinary React content, so props update them.
  return (
    <>
      <button onClick={onToggle}>{playing ? 'Pause' : 'Play'}</button>

      <BuoyLive
        label={`Now playing: ${track.title} by ${track.artist}`}
        expandable
      >
        {({ expanded }) =>
          expanded ? (
            <div className="buoy-exp">
              <span className="buoy-exp-title">{track.title}</span>
              <span className="buoy-exp-desc">{track.artist}</span>
              <div className="buoy-actions">
                <button className="buoy-btn is-primary" onClick={onToggle}>
                  {playing ? 'Pause' : 'Play'}
                </button>
              </div>
            </div>
          ) : (
            <span className="buoy-row no-icon">
              <span className="buoy-title">{track.title}</span>
            </span>
          )
        }
      </BuoyLive>
    </>
  );
}
