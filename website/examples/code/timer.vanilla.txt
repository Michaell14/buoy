import { buoy } from 'buoy-ui/vanilla';

const PILL =
  '<span class="buoy-row no-icon"><span class="buoy-title">Focus</span>' +
  '<span class="buoy-pct"></span></span>';
const clock = (s) =>
  `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;

function startTimer(minutes) {
  let left = minutes * 60;

  const timer = buoy.live({
    label: `Focus timer, ${minutes} minutes`,
    // Runs however the activity ends.
    onClose: () => clearInterval(tick),
    render(el) {
      el.innerHTML = PILL;
      el.querySelector('.buoy-pct').textContent = clock(left);
    },
  });

  const tick = setInterval(() => {
    left -= 1;
    // Draw again in place. Nothing moves or is announced, so
    // every second is fine.
    if (left > 0) return timer.refresh();
    timer.close();
    buoy.success('Focus session done');
  }, 1000);

  return timer;
}

document.querySelector('#start').onclick = () => startTimer(25);
