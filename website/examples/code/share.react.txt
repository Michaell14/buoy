import { buoy } from 'buoy-ui';

export function ShareLink({ url }: { url: string }) {
  async function copy() {
    try {
      await navigator.clipboard.writeText(url);
      // The id keeps this to one message, however often the
      // button is pressed.
      buoy('Link copied', { id: 'share', icon: 'link' });
    } catch {
      buoy.warn("Couldn't copy the link", { id: 'share' });
    }
  }

  return (
    <div className="share">
      <input readOnly value={url} aria-label="Share link" />
      <button onClick={copy}>Copy link</button>
    </div>
  );
}
