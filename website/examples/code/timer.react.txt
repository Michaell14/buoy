import { useEffect, useState } from 'react';
import { buoy, BuoyLive } from 'buoy-ui';

const clock = (s: number) =>
  `${Math.floor(s / 60)}:${String(s % 60).padStart(2, '0')}`;

export function FocusTimer({ minutes = 25 }) {
  const [left, setLeft] = useState(minutes * 60);
  const [running, setRunning] = useState(false);

  useEffect(() => {
    if (!running) return;
    const tick = setInterval(() => setLeft((s) => s - 1), 1000);
    return () => clearInterval(tick);
  }, [running]);

  // When the time runs out, an ordinary message takes over.
  useEffect(() => {
    if (left > 0) return;
    setRunning(false);
    setLeft(minutes * 60);
    buoy.success('Focus session done');
  }, [left, minutes]);

  return (
    <>
      <button onClick={() => setRunning(!running)}>
        {running ? 'Pause' : 'Start'} · {clock(left)}
      </button>

      {running && (
        // Each render draws again in place. Nothing is announced.
        <BuoyLive label={`Focus timer, ${minutes} minutes`}>
          <span className="buoy-row no-icon">
            <span className="buoy-title">Focus</span>
            <span className="buoy-pct">{clock(left)}</span>
          </span>
        </BuoyLive>
      )}
    </>
  );
}
