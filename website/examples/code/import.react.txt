import { buoy } from 'buoy-ui';

const count = new Intl.NumberFormat();

export function ImportButton({ file, onImported }: Props) {
  async function start() {
    const job = buoy.track(
      async (task) => {
        // 1 on the first run, 2 after a retry: pick up where
        // the last run stopped.
        const run = await openImport(file, {
          resume: task.attempt > 1,
        });
        for await (const step of run.steps(task.signal)) {
          await task.checkpoint();
          task.progress(step.done, step.total);
        }
        // Anything thrown in here marks the task as failed
        // and shows its message.
        return run.summary;
      },
      {
        label: file.name,
        icon: 'user',
        // The amount line reads in rows, not bytes.
        format: (done, total) =>
          `${count.format(done)} of ${count.format(total)} rows`,
        doneLabel: 'Imported just now',
      },
    );

    try {
      // Resolves with what the work returned, once it has
      // finished, however many tries that took.
      onImported(await job.done);
    } catch {
      // Canceled, or removed before it finished.
    }
  }

  return <button onClick={start}>Import contacts</button>;
}
