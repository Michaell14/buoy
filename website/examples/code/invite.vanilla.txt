import { buoy } from 'buoy-ui/vanilla';

const form = document.querySelector('#invite');
const members = ['ana@example.com'];

form.addEventListener('submit', (event) => {
  event.preventDefault();
  const email = form.elements.email.value.trim();
  // One id, so each result replaces the last one in place.
  if (!email.includes('@')) {
    buoy.warn('Enter a valid email address', { id: 'invite' });
  } else if (members.includes(email)) {
    buoy.info(`${email} is already on the team`, { id: 'invite' });
  } else {
    sendInvite(email);
    buoy.success(`Invite sent to ${email}`, { id: 'invite' });
    form.reset();
  }
});
