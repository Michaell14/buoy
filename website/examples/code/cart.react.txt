import { useState } from 'react';
import { buoy } from 'buoy-ui';

// Any trusted <svg> string works where a built-in icon name
// would go.
const BAG =
  '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor"' +
  ' stroke-width="2"><path d="M5.5 8.5h13l1 11h-15z"/>' +
  '<path d="M9 8.5a3 3 0 0 1 6 0"/></svg>';

export function AddToCart({ item }: { item: Item }) {
  const [count, setCount] = useState(0);

  function add() {
    const next = count + 1;
    setCount(next);
    buoy('Added to cart', {
      id: 'cart',
      icon: BAG,
      // One button on the pill. The message waits six seconds
      // for it.
      action: {
        label: 'View',
        onClick: () => showCart(item, next),
      },
    });
  }

  return <button onClick={add}>Add to cart</button>;
}

function showCart(item: Item, count: number) {
  // A description and two actions open the same message into
  // a card.

    id: 'cart',

      { label: 'Keep shopping' },

  });
}
